cmake_minimum_required(VERSION 3.20)
project(OBSYDCosmetics VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
string(RANDOM LENGTH 8 ALPHABET 0123456789abcdef OBSYD_BUILD_SEED)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
find_package(Threads REQUIRED)
add_executable(protocol-tests tests/protocol.cpp)
target_compile_definitions(protocol-tests PRIVATE OBSYD_STRING_SEED=0x${OBSYD_BUILD_SEED}u)
add_test(NAME protocol COMMAND protocol-tests)
add_executable(session-tests tests/session.cpp client/session_service.cpp)
target_compile_definitions(session-tests PRIVATE OBSYD_STRING_SEED=0x${OBSYD_BUILD_SEED}u)
target_link_libraries(session-tests PRIVATE Threads::Threads)
add_test(NAME session COMMAND session-tests)
add_executable(security-report-tests tests/security_report.cpp)
add_test(NAME security-report COMMAND security-report-tests)
add_executable(language-tests tests/language.cpp client/i18n.cpp)
target_link_libraries(language-tests PRIVATE Threads::Threads)
add_test(NAME language COMMAND language-tests)

if(WIN32)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/client/config.generated.hpp")
    message(FATAL_ERROR "Generate a local configuration with Build-Windows.cmd or follow docs/SELF-HOSTING.md.")
  endif()
  enable_language(RC)
  add_library(platform STATIC client/platform.cpp)
  target_compile_definitions(platform PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00 OBSYD_STRING_SEED=0x${OBSYD_BUILD_SEED}u)
  target_link_libraries(platform PUBLIC bcrypt crypt32 winhttp advapi32 shell32)
  find_package(Java 17 REQUIRED COMPONENTS Development)
  find_program(NODE_EXECUTABLE node REQUIRED)
  get_filename_component(JAVA_COMPILER_REAL "${Java_JAVAC_EXECUTABLE}" REALPATH)
  get_filename_component(JAVA_BIN "${JAVA_COMPILER_REAL}" DIRECTORY)
  get_filename_component(JAVA_ROOT "${JAVA_BIN}" DIRECTORY)
  find_path(JAVA_JNI_INCLUDE jni.h HINTS "$ENV{JAVA_HOME}/include" "${JAVA_ROOT}/include" REQUIRED)
  file(GLOB_RECURSE AGENT_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/agent/src/*.java")
  add_custom_target(java-agent
    COMMAND "${CMAKE_COMMAND}" -E env "JAVA_HOME=${JAVA_ROOT}" "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build-agent.mjs"
    BYPRODUCTS "${CMAKE_CURRENT_SOURCE_DIR}/dist/OBSYDAgent.jar" VERBATIM)
  add_library(OBSYDPermit SHARED agent/native/permit.cpp)
  target_include_directories(OBSYDPermit PRIVATE "${JAVA_JNI_INCLUDE}" "${JAVA_JNI_INCLUDE}/win32")
  target_link_libraries(OBSYDPermit PRIVATE platform)
  add_dependencies(java-agent OBSYDPermit)
  set_target_properties(OBSYDPermit PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/dist" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/dist")
  add_custom_target(component-manifest
    COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build-component-manifest.mjs"
    BYPRODUCTS "${CMAKE_CURRENT_SOURCE_DIR}/build/components.signed.txt" "${CMAKE_CURRENT_SOURCE_DIR}/build/obsyd-server.txt" VERBATIM)
  add_dependencies(component-manifest java-agent)
  add_executable(OBSYDCosmetics WIN32 client/main.cpp client/lunar.cpp client/i18n.cpp client/i18n_windows.cpp client/security.cpp client/session_service.cpp client/app.rc)
  set_source_files_properties(client/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/dist/OBSYDAgent.jar;${CMAKE_CURRENT_SOURCE_DIR}/dist/OBSYDPermit.dll;${CMAKE_CURRENT_SOURCE_DIR}/build/obsyd-server.txt;${CMAKE_CURRENT_SOURCE_DIR}/assets/THIRD_PARTY_NOTICES.txt;${CMAKE_CURRENT_SOURCE_DIR}/build/components.signed.txt")
  add_dependencies(OBSYDCosmetics component-manifest OBSYDPermit)
  target_include_directories(OBSYDCosmetics PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/client")
  target_link_libraries(OBSYDCosmetics PRIVATE platform gdi32 user32 wintrust Threads::Threads)
  add_executable(windows-tests tests/windows.cpp)
  target_link_libraries(windows-tests PRIVATE platform)
  add_test(NAME windows-platform COMMAND windows-tests)
  add_executable(security-windows-tests tests/security_windows.cpp client/security.cpp client/i18n.cpp client/app.rc)
  add_dependencies(security-windows-tests component-manifest)
  target_link_libraries(security-windows-tests PRIVATE platform wintrust)
  add_test(NAME windows-components COMMAND security-windows-tests)
  if(MSVC)
    target_link_options(OBSYDCosmetics PRIVATE /MANIFEST:NO)
    foreach(target platform OBSYDPermit OBSYDCosmetics windows-tests protocol-tests session-tests security-report-tests security-windows-tests language-tests)
      target_compile_options(${target} PRIVATE /W4 /utf-8 /GS /guard:cf $<$<COMPILE_LANGUAGE:CXX>:/constexpr:steps2000000> $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/GL>)
      target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> /INCREMENTAL:NO)
    endforeach()
  elseif(MINGW)
    target_link_options(OBSYDPermit PRIVATE -static -static-libgcc -static-libstdc++ -flto -s -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
    target_compile_options(OBSYDPermit PRIVATE -O2 -flto -fstack-protector-strong)
    target_link_options(OBSYDCosmetics PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
    target_link_options(windows-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(security-windows-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_compile_options(platform PRIVATE -O2 -flto -fstack-protector-strong)
    target_compile_options(OBSYDCosmetics PRIVATE -O2 -flto -fstack-protector-strong)
    target_link_options(OBSYDCosmetics PRIVATE -flto -s)
  endif()
endif()
